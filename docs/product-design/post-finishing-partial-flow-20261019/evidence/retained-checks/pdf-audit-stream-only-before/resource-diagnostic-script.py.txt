def canonical(value,seen=None):

  return ['object',hashed]
 if isinstance(value,StreamObject):return {'dictionary':{str(k):canonical(v,seen) for k,v in sorted(value.items()) if str(k) not in ['/Length','/Filter']},'decodedBytesSha256':sha(value.get_data())}
 if isinstance(value,DictionaryObject):return {str(k):canonical(v,seen) for k,v in sorted(value.items())}
 if isinstance(value,(ArrayObject,list,tuple)):return [canonical(v,seen) for v in value]
 if isinstance(value,ByteStringObject):return ['bytes',sha(bytes(value))]
 if isinstance(value,NullObject):return None
 if isinstance(value,BooleanObject):return bool(value.value)
 if isinstance(value,(int,float,str,bool)) or value is None:return value
 return str(value)
