 const controls=[...el.querySelectorAll('button,summary,select')].filter(visible).map(x=>({text:x.getAttribute('aria-label')||x.textContent.trim(),...rect(x),fontSize:getComputedStyle(x).fontSize}));
 const ys=[...new Set(controls.map(x=>Math.round(x.y)))], col=controls.find(x=>x.text==='列设置');

}
