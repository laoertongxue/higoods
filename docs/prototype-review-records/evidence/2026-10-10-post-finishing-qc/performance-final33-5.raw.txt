        samples.push({route:r.path,device,trial,kind:'navigation',ms:navigation,pass:navigation<=1000,diagnostics:await diagnostics(p)})
        if(r.path.includes('print?') && device===devices[0]) {
          const begin=await p.evaluate(()=>performance.now())

        }
        const handlerErrors=await p.evaluate(()=>window.pfHandlerErrors);errors.push(...handlerErrors.map(error=>`${r.path} handler: ${error}`))
      }catch(e){errors.push(`${r.path} ${device.width} trial ${trial}: ${e.message}`)}finally{await c.close();await checkpoint()}
    }
  }
  return {segment,browser:page.context().browser().version(),service:base,data:'静态 3 个生产单 / 15 次回货 / 75 条 SKU',samples,inventory,errors,passed:errors.length===0&&samples.every(s=>s.pass)}
}
