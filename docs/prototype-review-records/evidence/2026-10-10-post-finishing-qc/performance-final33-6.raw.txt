  const measured=async(p,kind)=>{
    await p.waitForFunction(kind=>window.pfMeasurements?.some(s=>s.kind===kind&&s.generation===window.pfMeasurementGeneration),kind)
    return p.evaluate(kind=>window.pfMeasurements.filter(s=>s.kind===kind).at(-1).ms,kind)
  }
  const diagnostics=async p=>p.evaluate(()=>({visibility:document.visibilityState,focus:document.hasFocus(),longtasks:window.pfLongTasks,resources:performance.getEntriesByType('resource').sort((a,b)=>b.duration-a.duration).slice(0,5).map(e=>({url:e.name,duration:e.duration,size:e.transferSize}))}))
  for(const r of selected){
    const devices=r.pda?[{width:390,height:844}]:[{width:1366,height:768},{width:1280,height:720},{width:1024,height:768}]
    for(const device of devices)for(let trial=1;trial<=5;trial++){
      const {c,p}=await create(device,r)
      try{
        await p.goto(base+r.path,{waitUntil:'domcontentloaded'});const cold=await measured(p,'load')
        samples.push({route:r.path,device,trial,kind:'cold',ms:cold,pass:cold<=1000,diagnostics:await diagnostics(p)})
        await p.reload({waitUntil:'domcontentloaded'});const refresh=await measured(p,'load')
        samples.push({route:r.path,device,trial,kind:'refresh',ms:refresh,pass:refresh<=1000,diagnostics:await diagnostics(p)})
        await p.evaluate(({path,route})=>{window.pfBegin('list-navigation',{...route,previousMain:document.querySelector('main')});history.pushState({},'',path);window.dispatchEvent(new PopStateEvent('popstate'))},{path:r.path===prefix+'qc-orders'?prefix+'qc-workbench':prefix+'qc-orders',route:{title:r.path===prefix+'qc-orders'?'执行质检':'质检清单'}})
        await measured(p,'list-navigation')
        await p.evaluate(({path,route})=>{window.pfBegin('navigation',{...route,previousMain:document.querySelector('main')});history.pushState({},'',path);window.dispatchEvent(new PopStateEvent('popstate'))},{path:r.path,route:r})
        const navigation=await measured(p,'navigation')
        samples.push({route:r.path,device,trial,kind:'navigation',ms:navigation,pass:navigation<=1000,diagnostics:await diagnostics(p)})
        if(r.path.includes('print?') && device===devices[0]) {
          const begin=await p.evaluate(()=>performance.now())
          await p.pdf({path:`/tmp/post-finishing-print-${segment}-${selected.indexOf(r)}-${trial}.pdf`,format:'A4',printBackground:true})
          const finish=await p.evaluate(()=>performance.now())
          samples.push({route:r.path,device,trial,kind:'pdf-generation',ms:finish-begin,pass:finish-begin<=1000})
        }
        if(trial===1&&device===devices[0]){
          inventory.push({route:r.path,controls:await p.evaluate(()=>[...document.querySelectorAll('main button,main a[data-nav],main input,main select,main textarea,[data-testid] button,[data-testid] input,[data-testid] select')].filter(e=>e.getBoundingClientRect().width&&e.getBoundingClientRect().height&&!e.disabled).map(e=>({tag:e.tagName,text:e.textContent.trim().slice(0,60),attrs:Object.fromEntries([...e.attributes].filter(a=>a.name.startsWith('data-')||a.name==='name').map(a=>[a.name,a.value]))})))})
          await p.screenshot({path:`/tmp/post-finishing-${segment}-${selected.indexOf(r)}-${device.width}.png`,fullPage:false})
        }
        // First interaction of each route and repeated trials includes default
        // quantity entry and large-image actions without changing data volume.
        const image=p.locator('main [data-post-finishing-action="full-flow-zoom-image"], [data-qc-spu-color-images] button').first()
        if(await image.count()){
          await image.click();
          const ms=await measured(p,'image');samples.push({route:r.path,device,trial,kind:'image',ms,pass:ms<=1000});await p.keyboard.press('Escape');const closeMs=await measured(p,'image-close-escape');samples.push({route:r.path,device,trial,kind:'image-close-escape',ms:closeMs,pass:closeMs<=1000})
        }
        const handlerErrors=await p.evaluate(()=>window.pfHandlerErrors);errors.push(...handlerErrors.map(error=>`${r.path} handler: ${error}`))
      }catch(e){errors.push(`${r.path} ${device.width} trial ${trial}: ${e.message}`)}finally{await c.close();await checkpoint()}
    }
  }
  return {segment,browser:page.context().browser().version(),service:base,data:'静态 3 个生产单 / 15 次回货 / 75 条 SKU',samples,inventory,errors,passed:errors.length===0&&samples.every(s=>s.pass)}
}
)(page);
